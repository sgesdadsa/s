## 💻 COMO USAR:

### 🏗️ IDE Builder - Criar Projeto

#### Via API:
```python
import requests

# Criar projeto C# WinForms
response = requests.post(
    "http://localhost:8000/api/ide/project/create",
    json={
        "name": "MeuApp",
        "language": "csharp",
        "type": "winforms"
    }
)

project = response.json()
project_id = project['id']
```

#### Via WebSocket:
```javascript
const ws = new WebSocket('ws://localhost:8000/ws/ide');

ws.send(JSON.stringify({
    type: 'create_project',
    data: {
        name: 'MeuApp',
        language: 'csharp',
        type: 'winforms'
    }
}));
```

### 🎨 Visual Designer - Criar Form

```python
# Criar form
form_response = requests.post(
    "http://localhost:8000/api/ide/designer/form/create",
    json={
        "name": "MainForm",
        "title": "Meu Aplicativo",
        "size": {"width": 800, "height": 600},
        "type": "winforms",
        "language": "csharp"
    }
)

form = form_response.json()
form_id = form['id']

# Adicionar Button
requests.post(
    f"http://localhost:8000/api/ide/designer/form/{form_id}/control/add",
    json={
        "type": "Button",
        "name": "btnSalvar",
        "properties": {"Text": "Salvar"},
        "position": {"x": 100, "y": 50},
        "size": {"width": 100, "height": 30}
    }
)

# Gerar código
code_response = requests.get(
    f"http://localhost:8000/api/ide/designer/form/{form_id}/generate"
)

code = code_response.json()['code']
# code['designer'] = MainForm.Designer.cs
# code['main'] = MainForm.cs
# code['program'] = Program.cs
```

### 🔨 Build para Executável

```python
# Build projeto
build_response = requests.post(
    f"http://localhost:8000/api/ide/project/{project_id}/build",
    json={"config": "Release"}
)

build_result = build_response.json()

if build_result['success']:
    exe_path = build_result['exe_path']
    print(f"Build sucesso! EXE em: {exe_path}")
```

### 🐛 Debug

```python
# Iniciar debug
debug_response = requests.post(
    "http://localhost:8000/api/ide/debug/start",
    json={"project_id": project_id}
)

session_id = debug_response.json()['session_id']

# Set breakpoint
requests.post(
    f"http://localhost:8000/api/ide/debug/{session_id}/breakpoint",
    json={"file": "Program.cs", "line": 15}
)
```

### 🏭 Production Build

```python
# Build de produção com specs
production_response = requests.post(
    "http://localhost:8000/api/ide/production/build",
    json={
        "doc_url": "https://docs.example.com/specs.json",
        # OU
        "credentials_file": "credentials.json",
        # Specs:
        "name": "AppProdução",
        "language": "csharp",
        "type": "winforms"
    }
)

result = production_response.json()

if result['success']:
    download_url = result['download_url']

    # Download ZIP com .exe pronto
    package = requests.get(f"http://localhost:8000{download_url}")

    with open('app_producao.zip', 'wb') as f:
        f.write(package.content)

    print("Package pronto para download!")
```
